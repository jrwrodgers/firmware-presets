#$ DESCRIPTION: AUX4 mid and high: BLACKBOX
#$ DESCRIPTION: AUX5 VTX power: low 25mW (power 1), mid 25mW (power 1), high power 3

#$ INCLUDE: presets/4.3/modes/defaults.txt

# -- Mode switches --
# ARM on AUX1 high
aux 0 0 0 1725 2100 0 0
# ANGLE on AUX3 mid
aux 1 1 2 1300 1700 0 0
# BEEPER on AUX2 high
aux 2 13 1 1750 2100 0 0
# BLACKBOX on AUX4 mid and high
aux 3 26 3 1300 2100 0 0
# FLIP OVER AFTER CRASH (turtle) on AUX2 high
aux 4 35 1 1725 2100 0 0

# -- Clear all VTX power ranges --
vtx 0 0 0 0 0 900 900
vtx 1 0 0 0 0 900 900
vtx 2 0 0 0 0 900 900
vtx 3 0 0 0 0 900 900
vtx 4 0 0 0 0 900 900
vtx 5 0 0 0 0 900 900
vtx 6 0 0 0 0 900 900
vtx 7 0 0 0 0 900 900
vtx 8 0 0 0 0 900 900
vtx 9 0 0 0 0 900 900

# -- AUX5 VTX power switch (uses existing VTX table levels) --
# low: power 1
vtx 0 4 0 0 1 900 1200
# mid: power 1
vtx 1 4 0 0 1 1300 1700
# high: power 3
vtx 2 4 0 0 3 1800 2100
